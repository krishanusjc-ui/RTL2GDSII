`timescale 1ns/1ps

module tb_digital_time_controller;

    parameter N = 8;

    reg             clk;
    reg             rst_n;
    reg             enable;
    reg             load;
    reg [N-1:0]     load_value;

    wire [N-1:0]    count;
    wire            timeout;

    // DUT
    digital_time_controller #(
        .N(N)
    ) dut (
        .clk(clk),
        .rst_n(rst_n),
        .enable(enable),
        .load(load),
        .load_value(load_value),
        .count(count),
        .timeout(timeout)
    );

    // Clock (10 ns period)
    always #5 clk = ~clk;

    initial begin
        $dumpfile("dump.vcd");      // VCD file name
        $dumpvars(0, tb_digital_time_controller);
        // Init
        clk        = 0;
        rst_n      = 0;
        enable     = 0;
        load       = 0;
        load_value = 0;

        // Reset
        #20 rst_n = 1;

        // Load 5
        #10;
        load       = 1;
        load_value = 8'd5;

        #10;
        load   = 0;
        enable = 1;

        // Let it count down
        #100;

        // Stop
        enable = 0;

        #20 $finish;
    end

    initial begin
        $display("Time | load enable | load_value | count | timeout");
        $monitor("%4t |  %b     %b    |    %0d     |   %0d  |   %b",
                 $time, load, enable, load_value, count, timeout);
    end

endmodule
